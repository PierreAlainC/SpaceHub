<?php

namespace App\Service;

use App\Entity\Planet;
use App\Repository\PlanetRepository;
use Doctrine\ORM\EntityManagerInterface;
use Symfony\Contracts\HttpClient\HttpClientInterface;

class PlanetService
{
    private HttpClientInterface $client;
    private string $apiKey;
    private string $apiUrl;
    private PlanetRepository $planetRepository;
    private EntityManagerInterface $entityManager;

    public function __construct(
        HttpClientInterface $client,
        string $solarSystemApiKey,
        string $solarSystemApiUrl,
        PlanetRepository $planetRepository,
        EntityManagerInterface $entityManager
    ) {
        $this->client = $client;
        $this->apiKey = $solarSystemApiKey;
        $this->apiUrl = $solarSystemApiUrl;
        $this->planetRepository = $planetRepository;
        $this->entityManager = $entityManager;
    }

    public function fetchPlanets(): array
    {
        $response = $this->client->request(
            'GET',
            $this->apiUrl . '/bodies',
            [
                'headers' => [
                    'Authorization' => 'Bearer ' . $this->apiKey,
                ],
            ]
        );

        $data = $response->toArray();

        return array_filter($data['bodies'], function ($body) {
            return isset($body['isPlanet']) && $body['isPlanet'] === true;
        });
    }

    public function updatePlanetsFromApi(): array
    {
        $apiPlanets = $this->fetchPlanets();

        $summary = [
            'updates' => 0,
            'skipped' => 0,
            'updated_planets' => [],
        ];

        foreach ($apiPlanets as $apiPlanet) {
            $planet = $this->planetRepository->findOneBy([
                'slug' => strtolower($apiPlanet['englishName'])
            ]);

            if (!$planet) {
                $summary['skipped']++;
                continue;
            }

            $currentMoonsCount = $planet->getMoonsCount() ?? 0;
            $moonsCount = isset($apiPlanet['moons']) && is_array($apiPlanet['moons']) ? count($apiPlanet['moons']) : 0;
            $discoveryDate = !empty($apiPlanet['discoveryDate']) ? $apiPlanet['discoveryDate'] : null;
            $updatedFields = [];

            // English Name
            if ($planet->getEnglishName() !== ($apiPlanet['englishName'] ?? null)) {
                $planet->setEnglishName($apiPlanet['englishName'] ?? null);
                $updatedFields[] = 'englishName';
            }

            // Mass
            $mass = $this->theMass($apiPlanet);
            if ($this->areMassesDifferent($planet->getMass(), $mass)) {
                $planet->setMass($mass);
                $updatedFields[] = 'mass';
            }

            // Diameter
            $diameter = $this->radiusToDiameter($apiPlanet);
            if ($this->areFloatsDifferent($planet->getDiameter(), $diameter)) {
                $planet->setDiameter($diameter);
                $updatedFields[] = 'diameter';
            }

            //Gravity
            $gravity = isset($apiPlanet['gravity']) ? $this->normalizeFloat((float) $apiPlanet['gravity']) : null;
            if ($this->areFloatsDifferent($planet->getGravity(), $gravity)) {
                $planet->setGravity($gravity);
                $updatedFields[] = 'gravity';
            }

            //Density
            $density = isset($apiPlanet['density']) ? $this->normalizeFloat((float) $apiPlanet['density']) : null;
            if ($this->areFloatsDifferent($planet->getDensity(), $density)) {
                $planet->setDensity($density);
                $updatedFields[] = 'density';
            }

            //Mean temperature
            $meanTemperature = isset($apiPlanet['avgTemp']) ? (int) $apiPlanet['avgTemp'] : null;
            if ($planet->getMeanTemperature() !== $meanTemperature) {
                $planet->setMeanTemperature($meanTemperature);
                $updatedFields[] = 'avg Temp';
            }

            //Semi major axis
            // Comparaison via méthode dédiée pour éviter les faux écarts de float / typage API
            $semimajorAxis = isset($apiPlanet['semimajorAxis']) ? $this->normalizeFloat((float) $apiPlanet['semimajorAxis']) : null;
            if ($this->areFloatsDifferent($planet->getSemimajorAxis(), $semimajorAxis)) {
                $planet->setSemimajorAxis($semimajorAxis);
                $updatedFields[] = 'semi major Axis';
            }

            //Orbital period
            $orbitalPeriod = isset($apiPlanet['sideralOrbit']) ? $this->normalizeFloat((float) $apiPlanet['sideralOrbit']) : null;
            if ($this->areFloatsDifferent($planet->getOrbitalPeriod(), $orbitalPeriod)) {
                $planet->setOrbitalPeriod($orbitalPeriod);
                $updatedFields[] = 'sideral Orbit';
            }

            //Moons count
            if ($currentMoonsCount !== $moonsCount) {
                $planet->setMoonsCount($moonsCount);
                $updatedFields[] = 'moons';
            }

            //Discovered by
            $discoveredBy = !empty($apiPlanet['discoveredBy']) ? $apiPlanet['discoveredBy'] : null;
            if ($planet->getDiscoveredBy() !== $discoveredBy) {
                $planet->setDiscoveredBy($discoveredBy);
                $updatedFields[] = 'discovered By';
            }

            //Discovered date
            if ($planet->getDiscoveredDate() !== $discoveryDate) {
                $planet->setDiscoveredDate($discoveryDate);
                $updatedFields[] = 'discovery Date';
            }

            /* Cela aurait pu être bien d'avoir la vitesse orbitale mais je n'ai pas cette information pour le moment */
            /* $planet->setOrbitalSpeed(null); */

            // Mise à jour timestamp seulement si modification
            if (!empty($updatedFields)) {
                $planet->setUpdatedAt(new \DateTimeImmutable());

                $summary['updates']++;

                $summary['updated_planets'][] = [
                    'name' => $planet->getName(),
                    'updated_fields' => $updatedFields,
                ];
            } else {
                $summary['skipped']++;
            }
        }

        $this->entityManager->flush();

        return $summary;
    }

    /* De manière à faire plus propre on prend la logique d'affichage de la masse à part */
    private function theMass(array $apiPlanet): ?float
    {
        if (!isset($apiPlanet['mass']['massValue'], $apiPlanet['mass']['massExponent'])) {
            return null;
        }

        $mass = (float) $apiPlanet['mass']['massValue'] * pow(10, (int) $apiPlanet['mass']['massExponent']);

        return $this->normalizeFloat($mass);
    }

    /* On a le rayon moyen d'une planète mais on veut afficher le diamètre */
    private function radiusToDiameter(array $apiPlanet): ?float
    {
        if (!isset($apiPlanet['meanRadius'])) {
            return null;
        }

        return $this->normalizeFloat((float) $apiPlanet['meanRadius'] * 2);
    }

    /**
     * Compare deux floats avec une tolérance pour éviter les fausses différences
     * liées aux imprécisions des nombres à virgule.
     */
    private function areFloatsDifferent(?float $a, ?float $b, float $epsilon = 0.0001): bool
    {
        if ($a === null && $b === null) {
            return false;
        }

        if ($a === null || $b === null) {
            return true;
        }

        return abs($a - $b) > $epsilon;
    }

    //Pour éviter les trop grands nombre qui peuvent poser problème sur la sync
    //avec des valeurs très légerment différentes à chaque fois
    private function normalizeFloat(?float $value, int $precision = 4): ?float
    {
        if ($value === null) {
            return null;
        }

        return round($value, $precision);
    }

    private function areMassesDifferent(?float $a, ?float $b, float $relativeTolerance = 0.00001): bool
    {
        if ($a === null && $b === null) {
            return false;
        }

        if ($a === null || $b === null) {
            return true;
        }

        $difference = abs($a - $b);
        $largest = max(abs($a), abs($b));

        if ($largest === 0.0) {
            return false;
        }

        return ($difference / $largest) > $relativeTolerance;
    }
}
